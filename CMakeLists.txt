cmake_minimum_required(VERSION 3.22)
project(Plate140 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(JUCE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../dependencies/JUCE" CACHE PATH "JUCE 8.0.6 checkout")
if(NOT EXISTS "${JUCE_PATH}/CMakeLists.txt")
 include(FetchContent)
 FetchContent_Declare(JUCE URL https://codeload.github.com/juce-framework/JUCE/tar.gz/51a8a6d7aeae7326956d747737ccf1575e61e209 URL_HASH SHA256=20b740351cc030086951d87614bc08b42d072f1e4db68db1ef1d988e66dbbdbe)
 FetchContent_MakeAvailable(JUCE)
else()
 add_subdirectory("${JUCE_PATH}" JUCE)
endif()
set(FORMATS VST3 Standalone)
if(APPLE)
 list(APPEND FORMATS AU)
endif()
juce_add_plugin(Plate140 COMPANY_NAME Aulamix PLUGIN_MANUFACTURER_CODE Aumx PLUGIN_CODE Pl14 FORMATS ${FORMATS} PRODUCT_NAME "PLATE 140" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_add_binary_data(PlateArtwork SOURCES docs/images/hardware-concept.png Source/GUI/Assets/PanelBackground.png)
juce_generate_juce_header(Plate140)
target_sources(Plate140 PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(Plate140 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Plate140 PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC PlateArtwork juce::juce_recommended_config_flags)
include(CTest)
if(BUILD_TESTING)
 add_executable(PlateTests Tests/TestPlateEngine.cpp)
 target_include_directories(PlateTests PRIVATE Source)
 add_test(NAME PlateDSP COMMAND PlateTests)
 add_executable(PlateAudibility Tests/MeasureAudibility.cpp)
 target_include_directories(PlateAudibility PRIVATE Source)
 juce_add_gui_app(PlatePreview PRODUCT_NAME PlatePreview)
 juce_generate_juce_header(PlatePreview)
 target_sources(PlatePreview PRIVATE Tests/Preview.cpp)
 target_include_directories(PlatePreview PRIVATE Source)
 target_link_libraries(PlatePreview PRIVATE Plate140 PlateArtwork juce::juce_audio_utils juce::juce_dsp)
endif()
